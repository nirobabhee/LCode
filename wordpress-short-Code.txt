2) Make Helper function for this action.

if (!function_exists('job_shortcode')) {
    function job_shortcode()
    {
        $jobs = JobPost::where('status', 1)->where('vacancy_available', '>', 0)->orderBy('id', 'desc')->paginate(microlab_paginate());
        ob_start();
        echo '<div class="viser-job-lists">';
        microlab_include('user/partials/jobs', ['jobs' => $jobs]);
        echo '</div>';
        return ob_get_clean();
    }
}
