@push('script')
    <script>
        "use strict";
        $("[name=name_sku]").on('keyup', function() {

            let data = {};
            data.name_sku = $(this).val();
            if ($(this).val().length < 1) {
                $(".products").addClass('d-none');
                return;
            } else {
                $(".products").removeClass('d-none');
            }

            $.ajax({
                url: "{{ route('admin.purchases.product.autocomplete') }}",
                method: 'GET',
                data: data,
                success: function(response) {
                    var products = '';
                    $(".products").html('');
                    $.each(response.data, function(key, product)
                    {
                        products += `<li class="ml-5 productItem"
                                        data-id="${product.id}"
                                        data-name="${product.name}"
                                        data-code="${product.code}"
                                        data-sku="${product.sku}"
                                        data-tax="${product.tax}"
                                        data-tax_amount="${product.tax_amount}"
                                        data-purchases_price="${product.purchases_price}"
                                        >
                                    ${product.name}</li>`;
                    });

                    $(".products").html(products);

                    ///start CLICK PRODUCT

                    $(".productItem").on('click', function() {
                        $("tbody").append(` <tr>
                                                <td data-label="@lang('Purchases Date')" class="fw-bold"> <input type="text" name="name" class="form-control" value="${$(this).data('name')}" readonly required> </td>
                                                <td data-label="@lang('Qty')"> <input type="text" name="quantity" value="1"  class="form-control" required></td>
                                                <td data-label="@lang('Rate')"> <input type="text" name="purchases_price" class="form-control" value="${$(this).data('purchases_price')}" required></td>
                                                <td data-label="@lang('Tax')"> <input type="text" name="tax_amount" class="form-control" value="${$(this).data('tax_amount')}" required></td>
                                                <td data-label="@lang('Total')"> <input type="text" name="total" class="form-control" required></td>
                                                <td data-label="@lang('Action')">
                                                    <div class="button--group">
                                                            <button class="btn btn-sm btn-outline--danger ms-1 removeBtn" onclick="$(this).parent().parent().parent().remove()" >
                                                                <i class="la la-trash"></i> @lang('Remove')
                                                            </button>
                                                    </div>
                                                </td>
                                            </tr>
                        `);
                        $(".products").addClass('d-none');
                        $('[name=name_sku]').val("");
                    });
                    ///End CLICK PRODUCT
                },
            });



        }).jQuery;
    </script>
@endpush




/////////////////////////////controller////////////////////////////////////
public function productAutocomplete(Request $request)
    {
        $products = Product::active()
            ->where('name', 'like', "%$request->name_sku%")
            ->orWhere('sku', 'like', "%$request->name_sku%")
            ->orWhere('code', 'like', "%$request->name_sku%")->get();
        if (!$products) {
            return response()->json([
                'success' => false,
                'message' => 'No data found'
            ]);
        } else {
            return response()->json([
                'success'    => true,
                'data'       => $products,
            ]);
        }
    }
