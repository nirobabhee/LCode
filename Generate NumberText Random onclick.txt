//Invoice-generate
            $(".generate").on('click', function() {
                function makeinvoiceno() {
                    var invoice = "";
                    var target = "0123456789";
                    for (var i = 0; i < 8; i++)
                        invoice += target.charAt(Math.floor(Math.random() * target.length));
                    return 'S' + invoice;
                }
                $("[name=invoice_no]").val(makeinvoiceno())
            });
