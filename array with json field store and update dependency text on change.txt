public function reviewSetting(){
        $pageTitle = 'Review Setting';
        return view('admin.setting.review', compact('pageTitle'));
    }
    
    public function reviewSettingStore(Request $request){
        $request->validate([
            'review' => 'required|array|size:5',
            'review.*' => 'required|string'
        ],[
            'review.*.required' => 'All field is required'
        ]); //this for custom validation msg .
    
        
        $general = GeneralSetting::first();
        
        $general->reviews = $request->review;
        $general->save();
        
        $notify[] = ['success','Updated successfully'];
        return back()->withNotify($notify);
        
    }
